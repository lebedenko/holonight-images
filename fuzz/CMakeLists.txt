if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "HOLONIGHT_IMAGES_FUZZING requires a separate Clang build")
endif()
add_library(images-fuzz-provider STATIC ../src/image.cpp ../src/metadata.cpp)
target_compile_features(images-fuzz-provider PUBLIC cxx_std_23)
target_include_directories(images-fuzz-provider PUBLIC ../include)
target_link_libraries(images-fuzz-provider PUBLIC Qt6::Core Qt6::Gui PkgConfig::EXIF PkgConfig::WebP)
target_compile_options(images-fuzz-provider PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
foreach(harness exif metadata decode)
  add_executable(fuzz-${harness} ${harness}.cpp)
  target_link_libraries(fuzz-${harness} PRIVATE images-fuzz-provider)
  target_compile_options(fuzz-${harness} PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
  target_link_options(fuzz-${harness} PRIVATE -fsanitize=fuzzer,address,undefined)
endforeach()
