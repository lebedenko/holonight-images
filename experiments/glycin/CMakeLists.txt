cmake_minimum_required(VERSION 3.25)
project(glycin-feasibility LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(PkgConfig REQUIRED)
pkg_check_modules(Glycin REQUIRED IMPORTED_TARGET glycin-2>=2.2)
add_executable(glycin-probe probe.cpp)
target_compile_features(glycin-probe PRIVATE cxx_std_23)
target_compile_definitions(glycin-probe PRIVATE QT_NO_KEYWORDS)
target_compile_options(glycin-probe PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(glycin-probe PRIVATE Qt6::Core Qt6::Gui PkgConfig::Glycin)
