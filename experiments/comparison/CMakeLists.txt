cmake_minimum_required(VERSION 3.25)
project(image-comparison LANGUAGES CXX)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Svg)
find_package(PkgConfig REQUIRED)
pkg_check_modules(EXIF REQUIRED IMPORTED_TARGET libexif)
pkg_check_modules(WebP REQUIRED IMPORTED_TARGET libwebp)
# Compile the actual provider sources without adding its install rules or changing exports.
add_library(comparison-provider STATIC ../../src/image.cpp ../../src/metadata.cpp ../../src/svg.cpp)
target_include_directories(comparison-provider PUBLIC ../../include)
target_compile_features(comparison-provider PUBLIC cxx_std_23)
target_link_libraries(comparison-provider PUBLIC Qt6::Core Qt6::Gui PRIVATE Qt6::Svg PkgConfig::EXIF PkgConfig::WebP)
add_executable(qt-worker qt-worker.cpp)
target_link_libraries(qt-worker PRIVATE comparison-provider Qt6::Svg)
target_compile_options(qt-worker PRIVATE -Wall -Wextra -Wpedantic)
add_executable(fault-worker fault-worker.cpp)
target_link_libraries(fault-worker PRIVATE Qt6::Core)
set(GLYCIN_BRIDGE "" CACHE FILEPATH "Private Rust static library built with cargo --release --locked")
if(GLYCIN_BRIDGE)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET gio-2.0)
  pkg_check_modules(LCMS REQUIRED IMPORTED_TARGET lcms2)
  pkg_check_modules(SECCOMP REQUIRED IMPORTED_TARGET libseccomp)
  pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
  add_executable(glycin-worker qt-worker.cpp)
  target_compile_definitions(glycin-worker PRIVATE WITH_GLYCIN)
  target_link_libraries(glycin-worker PRIVATE comparison-provider Qt6::Svg "${GLYCIN_BRIDGE}"
    PkgConfig::GLIB PkgConfig::LCMS PkgConfig::SECCOMP PkgConfig::FONTCONFIG dl pthread m)
endif()
