cmake_minimum_required(VERSION 3.25)
project(hash-correctness LANGUAGES CXX)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui)
add_executable(device-check device.cpp)
option(BUILD_VALIDATOR "Build archived validator" OFF)
if(BUILD_VALIDATOR)
  add_executable(validation-check validation.cpp ../../src/isolated/runtime.cpp
    ../../src/isolated/protocol.cpp ../../src/isolated/cleanup.cpp)
  target_include_directories(validation-check PRIVATE ../../include ../../src/isolated)
  target_compile_features(validation-check PRIVATE cxx_std_23)
  target_compile_options(validation-check PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(validation-check PRIVATE Qt6::Core Qt6::Gui)
endif()
target_compile_features(device-check PRIVATE cxx_std_23)
target_compile_options(device-check PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(device-check PRIVATE Qt6::Core)
