cmake_minimum_required(VERSION 3.25)
project(launcher-readiness-profile LANGUAGES CXX)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Svg)
set(HOLONIGHT_IMAGES_ISOLATED ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. images)
add_executable(launcher-readiness-profile harness.cpp)
target_include_directories(launcher-readiness-profile PRIVATE ../../src/isolated)
target_link_libraries(launcher-readiness-profile PRIVATE HolonightImages::Isolated)
target_compile_options(launcher-readiness-profile PRIVATE -Wall -Wextra -Wpedantic)
