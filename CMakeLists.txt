cmake_minimum_required(VERSION 3.25)
project(holonight-images VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui)
find_package(PkgConfig REQUIRED)
pkg_check_modules(EXIF REQUIRED IMPORTED_TARGET libexif)
pkg_check_modules(WebP REQUIRED IMPORTED_TARGET libwebp)
add_library(holonight_images STATIC src/image.cpp src/metadata.cpp)
add_library(HolonightImages::Images ALIAS holonight_images)
set_target_properties(holonight_images PROPERTIES EXPORT_NAME Images POSITION_INDEPENDENT_CODE ON)
target_compile_features(holonight_images PUBLIC cxx_std_23)
target_compile_options(holonight_images PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(holonight_images PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(holonight_images PUBLIC Qt6::Core Qt6::Gui PRIVATE PkgConfig::EXIF PkgConfig::WebP)
install(TARGETS holonight_images EXPORT HolonightImagesTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT HolonightImagesTargets NAMESPACE HolonightImages:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HolonightImages)
configure_package_config_file(cmake/HolonightImagesConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/HolonightImagesConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HolonightImages)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/HolonightImagesConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/HolonightImagesConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/HolonightImagesConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HolonightImages)
install(FILES LICENSE LICENSES/GPL-3.0-or-later.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_subdirectory(tests)
endif()
